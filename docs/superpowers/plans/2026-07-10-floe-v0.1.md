# floe v0.1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship floe v0.1 — a macOS CLI that turns iCloud Drive (or any synced folder) into a Time Machine-style versioned backup, powered by restic.

**Architecture:** Go CLI (cobra) orchestrating external tools (`restic`, `brctl`, `launchctl`, `security`, `osascript`) through a single `run.Runner` interface so every package is unit-testable with a fake runner. All commands are thin cobra wrappers over `internal/` packages. Spec: `docs/superpowers/specs/2026-07-10-floe-design.md`.

**Tech Stack:** Go ≥ 1.24, github.com/spf13/cobra, github.com/BurntSushi/toml. Everything else stdlib. Tests: stdlib `testing` only.

## Global Constraints

- Module path: `github.com/Jszigeti/floe`. macOS-only (darwin); CI runs on `macos-latest`.
- Dependencies limited to cobra + BurntSushi/toml. Fetch with `go get <pkg>@latest` (never guess versions).
- Commits: Conventional Commits, single line, with scope, imperative, lowercase, no trailing dot. **No body, no `Co-Authored-By` or any trailer. Never `--amend`, never `--no-verify`.**
- Code style: self-documenting names, comments only for non-obvious WHY (single-line, English), no dead code, no `any`-equivalent laziness (`interface{}` only where genuinely needed).
- All user-facing strings in English.
- Home-dir resolution ALWAYS via `os.UserHomeDir()` (reads `$HOME`, so tests isolate with `t.Setenv("HOME", t.TempDir())`).
- Every test command below is run from the repo root `~/floe`.

```go
package run

import (
	"bytes"
	"context"
	"fmt"
	"os"
	"os/exec"
)

type Cmd struct {
	Name string
	Args []string
	Env  []string
}

type Runner interface {
	Run(ctx context.Context, c Cmd) (string, error)
}

type ExecRunner struct{}

func (ExecRunner) Run(ctx context.Context, c Cmd) (string, error) {
	cmd := exec.CommandContext(ctx, c.Name, c.Args...)
	cmd.Env = append(os.Environ(), c.Env...)
	var stdout, stderr bytes.Buffer
	cmd.Stdout, cmd.Stderr = &stdout, &stderr
	if err := cmd.Run(); err != nil {
		return stdout.String(), fmt.Errorf("%s %v: %w: %s", c.Name, c.Args, err, stderr.String())
	}
	return stdout.String(), nil
}
```

`internal/run/fake.go`:
```go
package run

import (
	"context"
	"strings"
)

type Response struct {
	Out string
	Err error
}

type Fake struct {
	Calls     []Cmd
	Responses map[string]Response
	Handler   func(Cmd) (string, error)
	Default   Response
}

func (f *Fake) Run(_ context.Context, c Cmd) (string, error) {
	f.Calls = append(f.Calls, c)
	if f.Handler != nil {
		return f.Handler(c)
	}
	if r, ok := f.Responses[strings.TrimSpace(c.Name+" "+strings.Join(c.Args, " "))]; ok {
		return r.Out, r.Err
	}
	return f.Default.Out, f.Default.Err
}

// CalledWith reports whether any recorded call contains all given tokens in name+args.
func (f *Fake) CalledWith(tokens ...string) bool {
	for _, c := range f.Calls {
		joined := c.Name + " " + strings.Join(c.Args, " ")
		ok := true
		for _, tok := range tokens {
			if !strings.Contains(joined, tok) {
				ok = false
				break
			}
		}
		if ok {
			return true
		}
	}
	return false
}
```

```go
package config

import (
	"fmt"
	"os"
	"path/filepath"
	"strings"

	"github.com/BurntSushi/toml"
)

type Retention struct {
	Daily   int `toml:"daily"`
	Weekly  int `toml:"weekly"`
	Monthly int `toml:"monthly"`
}

type Scope struct {
	Paths   []string `toml:"paths"`
	Exclude []string `toml:"exclude"`
}

type Hooks struct {
	Pre  []string `toml:"pre"`
	Post []string `toml:"post"`
}

type Config struct {
	Destination string    `toml:"destination"`
	Hour        int       `toml:"hour"`
	Retention   Retention `toml:"retention"`
	Scope       Scope     `toml:"scope"`
	Hooks       Hooks     `toml:"hooks"`
}

func ICloudDrivePath(home string) string {
	return filepath.Join(home, "Library/Mobile Documents/com~apple~CloudDocs")
}

func DefaultExcludes(home string) []string {
	return []string{
		filepath.Join(home, "Library/Mobile Documents"),
		filepath.Join(home, "Library/CloudStorage"),
		filepath.Join(home, "Library/Caches"),
		filepath.Join(home, "Library/Containers/com.docker.docker"),
		filepath.Join(home, "Library/Developer"),
		filepath.Join(home, "Library/Application Support/MobileSync"),
		filepath.Join(home, ".Trash"),
		filepath.Join(home, ".cache"),
		filepath.Join(home, ".npm"),
		filepath.Join(home, ".nvm"),
		filepath.Join(home, ".cargo"),
		filepath.Join(home, ".rustup"),
		filepath.Join(home, ".gradle"),
		"node_modules",
		".venv",
		"venv",
		"__pycache__",
		".next",
		"dist",
		"target",
	}
}

func Default() (Config, error) {
	home, err := os.UserHomeDir()
	if err != nil {
		return Config{}, err
	}
	return Config{
		Destination: ICloudDrivePath(home),
		Hour:        13,
		Retention:   Retention{Daily: 7, Weekly: 4, Monthly: 6},
		Scope:       Scope{Paths: []string{home}, Exclude: DefaultExcludes(home)},
	}, nil
}

func Path() (string, error) {
	home, err := os.UserHomeDir()
	if err != nil {
		return "", err
	}
	return filepath.Join(home, ".config/floe/config.toml"), nil
}

func Load() (Config, error) {
	p, err := Path()
	if err != nil {
		return Config{}, err
	}
	var c Config
	if _, err := toml.DecodeFile(p, &c); err != nil {
		if os.IsNotExist(err) {
			return Config{}, fmt.Errorf("no config at %s — run `floe init` first", p)
		}
		return Config{}, fmt.Errorf("reading %s: %w", p, err)
	}
	return c, nil
}

func (c Config) Save() error {
	p, err := Path()
	if err != nil {
		return err
	}
	if err := os.MkdirAll(filepath.Dir(p), 0o755); err != nil {
		return err
	}
	f, err := os.Create(p)
	if err != nil {
		return err
	}
	defer f.Close()
	return toml.NewEncoder(f).Encode(c)
}

func (c Config) RepoPath() (string, error) {
	host, err := os.Hostname()
	if err != nil {
		return "", err
	}
	host = strings.TrimSuffix(host, ".local")
	host = strings.ReplaceAll(host, " ", "-")
	return filepath.Join(c.Destination, "Floe.backup", host), nil
}
```

```go
package keychain

import (
	"context"
	"crypto/rand"
	"encoding/base64"
	"fmt"
	"strings"

	"github.com/Jszigeti/floe/internal/run"
)

const (
	service = "floe"
	account = "restic"
)

func Generate() (string, error) {
	b := make([]byte, 24)
	if _, err := rand.Read(b); err != nil {
		return "", err
	}
	return base64.RawURLEncoding.EncodeToString(b), nil
}

func Set(ctx context.Context, r run.Runner, pass string) error {
	_, err := r.Run(ctx, run.Cmd{Name: "security", Args: []string{
		"add-generic-password", "-U", "-s", service, "-a", account, "-w", pass,
	}})
	return err
}

func Get(ctx context.Context, r run.Runner) (string, error) {
	out, err := r.Run(ctx, run.Cmd{Name: "security", Args: []string{
		"find-generic-password", "-s", service, "-a", account, "-w",
	}})
	if err != nil {
		return "", fmt.Errorf("no floe passphrase in keychain — run `floe init` first: %w", err)
	}
	return strings.TrimSpace(out), nil
}
```

```go
package restic

import (
	"compress/bzip2"
	"context"
	"crypto/sha256"
	"encoding/hex"
	"fmt"
	"io"
	"net/http"
	"os"
	"path/filepath"
	"runtime"
)

const Version = "0.18.1" // pinned; update via Step 1 of the download task

var checksums = map[string]string{
	"arm64": "<sha256 of restic_0.18.1_darwin_arm64.bz2 — paste from SHA256SUMS>",
	"amd64": "<sha256 of restic_0.18.1_darwin_amd64.bz2 — paste from SHA256SUMS>",
}

const defaultBaseURL = "https://github.com/restic/restic/releases/download/v" + Version

func BinPath() (string, error) {
	home, err := os.UserHomeDir()
	if err != nil {
		return "", err
	}
	return filepath.Join(home, ".local/share/floe/bin/restic"), nil
}

func EnsureBinary(ctx context.Context, baseURL string) (string, error) {
	dst, err := BinPath()
	if err != nil {
		return "", err
	}
	if _, err := os.Stat(dst); err == nil {
		return dst, nil
	}
	if baseURL == "" {
		baseURL = defaultBaseURL
	}
	want, ok := checksums[runtime.GOARCH]
	if !ok {
		return "", fmt.Errorf("unsupported architecture %s", runtime.GOARCH)
	}
	url := fmt.Sprintf("%s/restic_%s_darwin_%s.bz2", baseURL, Version, runtime.GOARCH)
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, nil)
	if err != nil {
		return "", err
	}
	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return "", fmt.Errorf("downloading restic: %w", err)
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		return "", fmt.Errorf("downloading restic: HTTP %d from %s", resp.StatusCode, url)
	}
	compressed, err := io.ReadAll(resp.Body)
	if err != nil {
		return "", err
	}
	sum := sha256.Sum256(compressed)
	if hex.EncodeToString(sum[:]) != want {
		return "", fmt.Errorf("restic checksum mismatch for %s", url)
	}
	binary, err := io.ReadAll(bzip2.NewReader(bytesReader(compressed)))
	if err != nil {
		return "", fmt.Errorf("decompressing restic: %w", err)
	}
	if err := os.MkdirAll(filepath.Dir(dst), 0o755); err != nil {
		return "", err
	}
	if err := os.WriteFile(dst, binary, 0o755); err != nil {
		return "", err
	}
	return dst, nil
}
```
(`bytesReader` = `bytes.NewReader`; import `bytes` and use it directly — the alias above is illustrative, write `bytes.NewReader(compressed)`.)

```go
import (
	"context"
	"io/fs"
	"path/filepath"
	"syscall"
	"time"

	"github.com/Jszigeti/floe/internal/run"

const sfDataless = 0x40000000

type FlagsFn func(path string) (uint32, error)

func StatFlags(path string) (uint32, error) {
	var st syscall.Stat_t
	if err := syscall.Stat(path, &st); err != nil {
		return 0, err
	}
	return st.Flags, nil
}

func DatalessFiles(root string, fn FlagsFn) ([]string, error) {
	var out []string
	err := filepath.WalkDir(root, func(p string, d fs.DirEntry, err error) error {
		if err != nil || d.IsDir() {
			return err
		}
		flags, ferr := fn(p)
		if ferr == nil && flags&sfDataless != 0 {
			out = append(out, p)
		}
		return nil
	})
	return out, err
}

func Materialize(ctx context.Context, r run.Runner, fn FlagsFn, paths []string) error {
	for _, p := range paths {
		if _, err := r.Run(ctx, run.Cmd{Name: "brctl", Args: []string{"download", p}}); err != nil {
			return err
		}
	}
	for {
		remaining := 0
		for _, p := range paths {
			if flags, err := fn(p); err == nil && flags&sfDataless != 0 {
				remaining++
			}
		}
		if remaining == 0 {
			return nil
		}
		select {
		case <-ctx.Done():
			return ctx.Err()
		case <-time.After(500 * time.Millisecond):
		}
	}
}
```

```go
package scan

import (
	"io/fs"
	"path/filepath"
	"sort"
	"strings"
)

type DirSize struct {
	Path  string
	Bytes int64
}

func Excluded(path string, excludes []string) bool {
	base := filepath.Base(path)
	for _, e := range excludes {
		if strings.ContainsRune(e, '/') {
			if path == e || strings.HasPrefix(path, e+"/") {
				return true
			}
		} else if base == e {
			return true
		}
	}
	return false
}

func Estimate(root string, excludes []string) (int64, []DirSize, error) {
	var total int64
	perChild := map[string]int64{}
	err := filepath.WalkDir(root, func(p string, d fs.DirEntry, err error) error {
		if err != nil {
			return nil // unreadable entries are skipped, not fatal
		}
		if Excluded(p, excludes) {
			if d.IsDir() {
				return filepath.SkipDir
			}
			return nil
		}
		if d.IsDir() {
			return nil
		}
		info, ierr := d.Info()
		if ierr != nil {
			return nil
		}
		total += info.Size()
		if rel, rerr := filepath.Rel(root, p); rerr == nil {
			child := strings.SplitN(rel, string(filepath.Separator), 2)[0]
			if child != rel { // file is inside a first-level directory
				perChild[filepath.Join(root, child)] += info.Size()
			}
		}
		return nil
	})
	if err != nil {
		return 0, nil, err
	}
	top := make([]DirSize, 0, len(perChild))
	for p, b := range perChild {
		top = append(top, DirSize{Path: p, Bytes: b})
	}
	sort.Slice(top, func(i, j int) bool { return top[i].Bytes > top[j].Bytes })
	return total, top, nil
}
```

```go
package backup

import (
	"context"
	"fmt"
	"time"

	"github.com/Jszigeti/floe/internal/config"
	"github.com/Jszigeti/floe/internal/digest"
	"github.com/Jszigeti/floe/internal/icloud"
	"github.com/Jszigeti/floe/internal/restic"
	"github.com/Jszigeti/floe/internal/run"
)

const lowQuotaBytes = 5 << 30

type Deps struct {
	Cfg    config.Config
	Client *restic.Client
	Runner run.Runner
	Notify func(title, msg string)
	Now    func() time.Time
	Log    func(format string, a ...any)
}

func Run(ctx context.Context, d Deps) error {
	if err := runHooks(ctx, d, d.Cfg.Hooks.Pre, "pre-hook"); err != nil {
		return err
	}

	onICloud := icloud.IsICloudPath(d.Cfg.Destination)
	if onICloud {
		if rem, err := icloud.QuotaRemaining(ctx, d.Runner); err == nil && rem < lowQuotaBytes {
			d.Notify("floe: iCloud quota low",
				fmt.Sprintf("%.1f GB remaining — reduce retention, add excludes, or upgrade iCloud+", float64(rem)/1e9))
		}
	}

	d.Client.Unlock(ctx) // removes stale locks only; failure is non-fatal
	excludes := append([]string{d.Cfg.Destination}, d.Cfg.Scope.Exclude...)
	if err := d.Client.Backup(ctx, d.Cfg.Scope.Paths, excludes); err != nil {
		d.Notify("floe backup failed", err.Error())
		return err
	}

	skipPrune := false
	if onICloud {
		if pending, err := icloud.SyncPending(ctx, d.Runner); err == nil && pending {
			skipPrune = true
			d.Log("repo sync pending, skipping prune this run")
		}
	}
	if !skipPrune {
		if err := d.Client.Forget(ctx, d.Cfg.Retention); err != nil {
			d.Notify("floe retention failed", err.Error())
			return err
		}
	}

	if err := runHooks(ctx, d, d.Cfg.Hooks.Post, "post-hook"); err != nil {
		return err
	}

	if digest.Due(d.Now()) {
		msg := "Backups healthy"
		if size, err := d.Client.RepoSize(ctx); err == nil {
			msg = fmt.Sprintf("Backups healthy — repo %.1f GB", float64(size)/1e9)
		}
		d.Notify("floe: backups healthy", msg)
		digest.MarkSent(d.Now())
	}
	return nil
}

func runHooks(ctx context.Context, d Deps, hooks []string, kind string) error {
	for _, h := range hooks {
		if _, err := d.Runner.Run(ctx, run.Cmd{Name: "sh", Args: []string{"-c", h}}); err != nil {
			d.Notify("floe backup failed", fmt.Sprintf("%s failed: %s", kind, h))
			return fmt.Errorf("%s %q: %w", kind, h, err)
		}
	}
	return nil
}
```

```go
			in := bufio.NewScanner(cmd.InOrStdin())
			ask := func(prompt, def string) string {

			}

						cfg.Scope.Exclude = append(cfg.Scope.Exclude, top[i-1].Path)

				pass, err = keychain.Generate()
				if err != nil {
					return err
				}
				if err := keychain.Set(ctx, d.Runner, pass); err != nil {
					return err
				}
				cmd.Println("\n════════════════════════════════════════════════")
				cmd.Printf("  Your backup passphrase: %s\n", pass)
				cmd.Println("  Save it in your password manager NOW.")
				cmd.Println("  Without it your backup is unreadable.")

			return nil, cfg, fmt.Errorf("initializing repository: %w", err)
		}
	}
	if err := cfg.Save(); err != nil {
		return nil, cfg, err
	}
	return client, cfg, nil
}
```

```go
package cli

import (
	"context"
	"fmt"
	"os"

	"github.com/Jszigeti/floe/internal/config"
	"github.com/Jszigeti/floe/internal/icloud"
	"github.com/Jszigeti/floe/internal/keychain"
	"github.com/Jszigeti/floe/internal/launchd"
	"github.com/Jszigeti/floe/internal/restic"
	"github.com/spf13/cobra"
)

type check struct {
	name string
	run  func(ctx context.Context) (ok bool, warn bool, msg string)
}

func newDoctorCmd(d Deps) *cobra.Command {
	return &cobra.Command{
		Use:   "doctor",
		Short: "Diagnose backup health and iCloud pitfalls",
		RunE: func(cmd *cobra.Command, _ []string) error {
			ctx := cmd.Context()
			cfg, err := config.Load()
			if err != nil {
				return err
			}
			repo, err := cfg.RepoPath()
			if err != nil {
				return err
			}
			checks := []check{
				{"restic binary", func(ctx context.Context) (bool, bool, string) {
					bin, err := restic.BinPath()
					if err != nil {
						return false, false, err.Error()
					}
					if _, err := os.Stat(bin); err != nil {
						return false, false, "missing — run `floe init`"
					}
					return true, false, bin
				}},
				{"keychain passphrase", func(ctx context.Context) (bool, bool, string) {
					if _, err := keychain.Get(ctx, d.Runner); err != nil {
						return false, false, "missing — run `floe init`"
					}
					return true, false, "present"
				}},
				{"schedule", func(ctx context.Context) (bool, bool, string) {
					if !launchd.IsLoaded(ctx, d.Runner) {
						return false, false, "launchagent not loaded — run `floe init`"
					}
					return true, false, fmt.Sprintf("daily at %02d:00", cfg.Hour)
				}},
				{"repository", func(ctx context.Context) (bool, bool, string) {
					client, _, err := buildClient(ctx, d.Runner)
					if err != nil {
						return false, false, err.Error()
					}
					if !client.IsInitialized(ctx) {
						return false, false, "unreachable or not initialized"
					}
					return true, false, repo
				}},
			}
			if icloud.IsICloudPath(cfg.Destination) {
				checks = append(checks,
					check{"optimize mac storage", func(ctx context.Context) (bool, bool, string) {
						if icloud.OptimizeStorageEnabled(ctx, d.Runner) {
							return true, true, "ON — macOS may evict repo files; disable it in System Settings → Apple ID → iCloud"
						}
						return true, false, "off"
					}},
					check{"repo files materialized", func(ctx context.Context) (bool, bool, string) {
						dataless, err := icloud.DatalessFiles(repo, icloud.StatFlags)
						if err != nil || len(dataless) == 0 {
							return true, false, "all local"
						}
						if err := icloud.Materialize(ctx, d.Runner, icloud.StatFlags, dataless); err != nil {
							return false, false, fmt.Sprintf("%d evicted files, re-download failed: %v", len(dataless), err)
						}
						return true, true, fmt.Sprintf("re-downloaded %d evicted files", len(dataless))
					}},
					check{"icloud quota", func(ctx context.Context) (bool, bool, string) {
						rem, err := icloud.QuotaRemaining(ctx, d.Runner)
						if err != nil {
							return true, true, "unknown"
						}
						if rem < 5<<30 {
							return true, true, fmt.Sprintf("only %.1f GB remaining", float64(rem)/1e9)
						}
						return true, false, fmt.Sprintf("%.1f GB remaining", float64(rem)/1e9)
					}},
				)
			}
			failed := false
			for _, c := range checks {
				ok, warn, msg := c.run(ctx)
				marker := "✓"
				if warn {
					marker = "!"
				}
				if !ok {
					marker, failed = "✗", true
				}
				cmd.Printf("%s %-26s %s\n", marker, c.name, msg)
			}
			if failed {
				return fmt.Errorf("doctor found problems")
			}
			return nil
		},
	}
}
```

```go
func TestFullBackupRestoreCycle(t *testing.T) {

	src := filepath.Join(home, "data")

	repo := filepath.Join(home, "dest/repo")
	os.MkdirAll(repo, 0o755)

	client := &restic.Client{Bin: bin, Repo: repo, Pass: "integration-pw", R: run.ExecRunner{}}

		t.Fatal(err)
	}

		Retention:   config.Retention{Daily: 7, Weekly: 4, Monthly: 6},
		Scope:       config.Scope{Paths: []string{src}},
	}

		Cfg: cfg, Client: client, Runner: run.ExecRunner{},
		Notify: func(string, string) {}, Now: time.Now, Log: t.Logf,
	}
	if err := backup.Run(ctx, d); err != nil {
		t.Fatal(err)
	}
	os.WriteFile(filepath.Join(src, "keep.txt"), []byte("v2"), 0o644)
	if err := backup.Run(ctx, d); err != nil {
		t.Fatal(err)
	}

	snaps, err := client.Snapshots(ctx)
	if err != nil || len(snaps) != 2 {
		t.Fatalf("snapshots = %d (%v), want 2", len(snaps), err)

	target := filepath.Join(home, "restored")

		t.Fatal(err)
	}

	if err != nil || string(got) != "v1" {

	}
```

```bash
TMP=$(mktemp -d)
trap 'rm -rf "$TMP"' EXIT

mkdir -p "$INSTALL_DIR"
mv "$TMP/floe" "$INSTALL_DIR/floe"
chmod +x "$INSTALL_DIR/floe"

echo "✓ floe $TAG installed to $INSTALL_DIR/floe"
case ":$PATH:" in
  *":$INSTALL_DIR:"*) ;;
  *) echo "Add it to your PATH: export PATH=\"$INSTALL_DIR:\$PATH\"" ;;

echo "Get started: floe init"
```

```bash
#!/bin/sh
# floe pre-hook example: export named Docker volumes as uncompressed tars
# (uncompressed so restic dedup works across days). Add to config.toml:
#   [hooks]
#   pre = ["~/.config/floe/hooks/export-docker-volumes.sh"]
set -eu
STAGING="$HOME/.local/state/floe/docker-volumes"
mkdir -p "$STAGING"
docker info >/dev/null 2>&1 || { echo "docker not running, skipping volume export"; exit 0; }
for v in $(docker volume ls -q); do
  docker run --rm -v "$v":/data:ro alpine tar cC /data . > "$STAGING/$v.tar"
done
echo "exported $(docker volume ls -q | wc -l | tr -d ' ') volumes to $STAGING"
```
